    What is KML?

    KML files are file formats specifically for storing and displaying geographic data in an Earth browser such as Google Earth.

    Since Google Earth was the tool recommended that was the one I used to view the .kml files but there are alternatives that can be used if you wish to research those resources.

    import csv
import simplekml

# input and output file paths
input_csv = "Preparations-drone-name.csv" # input the csv that needs to be converted
output_kml = "drone-name-out.kml" # output name of your choice 

# create new KML file
kml = simplekml.Kml()

# read the CSV file and add points
with open(input_csv, newline='', encoding='utf-8') as csvfile:
    reader = csv.DictReader(csvfile)
    for idx, row in enumerate(reader, start=1):
        lat = float(row["OSD.latitude"])
        lon = float(row["OSD.longitude"])
        kml.newpoint(name=f"Location {idx}", coords=[(lon, lat)])

# save the kml file
kml.save(output_kml)
print(f"KML file saved as {output_kml}")

    import csv
import matplotlib.pyplot as plt

input_csv = "ELF-HAWK-dump.csv"

# Initialize lists for latitudes and longitudes
lats = []
lons = []

# Read the CSV file
with open(input_csv, newline='', encoding='utf-8') as csvfile:
    reader = csv.DictReader(csvfile)
    for row in reader:
        try:
            lat = float(row["OSD.latitude"])
            lon = float(row["OSD.longitude"])

            # Ignore (0, 0) points
            if lat == 0 and lon == 0:
                continue

            lats.append(lat)
            lons.append(lon)
        except (ValueError, KeyError) as e:
            print(f"Error processing row: {row}, {e}")

# Plot the points
plt.figure(figsize=(10, 10))
plt.plot(lons, lats, linestyle='-', marker='', color='red')  # Red line connecting points
plt.scatter(lons, lats, s=1, c="blue")  # Optional: Show points as well
plt.title("LineStrings Visualization")
plt.xlabel("Longitude")
plt.ylabel("Latitude")
plt.grid(True)
plt.show()

problem_points = [(lon, lat) for lon, lat in zip(lons, lats) if lon == 0 and lat == 0]
if problem_points:
    problem_lons, problem_lats = zip(*problem_points)
    plt.scatter(problem_lons, problem_lats, s=20, c="red", label="Problem Points")
    plt.legend()
